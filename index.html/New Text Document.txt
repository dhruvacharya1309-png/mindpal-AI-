    async function initAudioEngine() {
      if (!audioContext) audioContext = new (window.AudioContext || window.webkitAudioContext)();
      if (audioContext.state === 'suspended') await audioContext.resume();
      if (!micStream) {
        micStream = await navigator.mediaDevices.getUserMedia({ audio: true });
        const source = audioContext.createMediaStreamSource(micStream);
        analyserNode = audioContext.createAnalyser();
        analyserNode.fftSize = 2048;
        source.connect(analyserNode);
      }
    }

    function drawWaveform(canvasId) {
      const canvas = document.getElementById(canvasId);
      if (!canvas || !analyserNode) return;
      const ctx = canvas.getContext('2d');
      const width = canvas.width = canvas.parentElement.clientWidth || 300;
      const height = canvas.height = 110;
      const bufferLength = analyserNode.frequencyBinCount;
      const dataArray = new Uint8Array(bufferLength);

      function render() {
        analyserNode.getByteTimeDomainData(dataArray);
        ctx.fillStyle = '#06080e';
        ctx.fillRect(0, 0, width, height);
        ctx.lineWidth = 2;
        ctx.strokeStyle = '#e5a93c';
        ctx.beginPath();
        const sliceWidth = width / bufferLength;
        let x = 0;
        for (let i = 0; i < bufferLength; i++) {
          const v = dataArray[i] / 128.0;
          const y = (v * height) / 2;
          if (i === 0) ctx.moveTo(x, y);
          else ctx.lineTo(x, y);
          x += sliceWidth;
        }
        ctx.lineTo(width, height / 2);
        ctx.stroke();
        requestAnimationFrame(render);
      }
      render();
    }

    function speakText(text) {
      return new Promise((resolve) => {
        if (!('speechSynthesis' in window)) return resolve();
        window.speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.rate = 1.02;
        utterance.pitch = 0.98;
        const voices = window.speechSynthesis.getVoices();
        const preferred = voices.find((v) => v.name.includes('Daniel') || v.name.includes('Natural')) || voices[0];
        if (preferred) utterance.voice = preferred;
        utterance.onend = () => resolve();
        utterance.onerror = () => resolve();
        window.speechSynthesis.speak(utterance);
      });
    }

    }

    function initSpeechRecognition() {
      const SpeechRec = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (!SpeechRec) return;
      speechRecognition = new SpeechRec();
      speechRecognition.continuous = true;
      speechRecognition.interimResults = true;
      speechRecognition.onresult = (event) => {
        let t = '';
        for (let i = 0; i < event.results.length; i++) t += event.results[i][0].transcript + ' ';
        currentLiveTranscript = t.trim();
        document.getElementById('transcript-stream').textContent = currentLiveTranscript;
      };
    }

    function runThinkingWindow(sec, allowScratchpad) {
      return new Promise((resolve) => {
        const banner = document.getElementById('thinking-banner');
        const count = document.getElementById('thinking-countdown');
        const hint = document.getElementById('scratchpad-hint');

        banner.style.display = 'block';
        hint.style.display = allowScratchpad ? 'block' : 'none';

        let rem = sec;
        count.textContent = rem;

        const itv = setInterval(() => {
          rem--;
          count.textContent = rem;
          if (rem <= 0) {
            clearInterval(itv);
            banner.style.display = 'none';
            resolve();
          }
        }, 1000);
      });
    }

    function startLiveRecording() {
      state.roundStartTime = Date.now();
      document.getElementById('rec-status-indicator').innerHTML = '<span style="color: var(--accent-rose);">● Live Recording</span>';

      audioChunks = [];
      try {
        mediaRecorder = new MediaRecorder(micStream, { mimeType: 'audio/webm' });
      } catch (e) {
        mediaRecorder = new MediaRecorder(micStream);
      }
      mediaRecorder.ondataavailable = (e) => { if (e.data.size > 0) audioChunks.push(e.data); };
      mediaRecorder.start();

      if (speechRecognition) {
        try { speechRecognition.start(); } catch (e) {}
      }

      clearInterval(roundTimer);
      roundTimer = setInterval(async () => {
        const elapsed = (Date.now() - state.roundStartTime) / 1000;
        if (elapsed >= 90 && !softWarningTriggered) {
          softWarningTriggered = true;
          document.getElementById('time-warning-banner').style.display = 'block';
        }
        if (elapsed >= 120) {
          clearInterval(roundTimer);
          document.getElementById('time-warning-banner').style.display = 'none';
          await speakText("Understood, we have enough on that core direction. Let's move into the operational specifics.");
          submitCurrentAnswer();
        }
      }, 1000);
    }

    async function stopLiveRecording() {
      clearInterval(roundTimer);
      document.getElementById('time-warning-banner').style.display = 'none';
      document.getElementById('rec-status-indicator').innerHTML = '<span>● Standby</span>';
      if (speechRecognition) {
        try { speechRecognition.stop(); } catch (e) {}
      }

      return new Promise((resolve) => {
        if (!mediaRecorder || mediaRecorder.state === 'inactive') return resolve(null);
        mediaRecorder.onstop = () => resolve(new Blob(audioChunks, { type: 'audio/webm' }));
        mediaRecorder.stop();
      });
    }

    async function submitCurrentAnswer() {
      const blob = await stopLiveRecording();
      const rawText = currentLiveTranscript.trim();
      const wordCount = rawText.split(/\s+/).filter(Boolean).length;

      if (wordCount === 0) {
        await speakText("No spoken answer was detected. In a placement drive, silence is recorded as an unattempted response. Moving to the next question.");
        logRoundTelemetry('[UNATTEMPTED - SILENCE]', 0, 0, 0, blob);
        advanceNextRound();
        return;
      }

      const durationMin = Math.max((Date.now() - state.roundStartTime) / 60000, 0.1);
      const wpm = Math.round(wordCount / durationMin);
      const fillers = (rawText.match(/\b(um|uh|basically|like|actually)\b/gi) || []).length;
      const currentQ = state.selectedQuestions[state.currentRoundIndex];

      let socratic = null;
      try {
        const res = await fetch('/api/interview', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            action: 'followup',
            payload: {
              question: currentQ.question,
              headEvaluationKey: currentQ.headEvaluationKey,
              candidateAnswer: rawText,
              domain: profile.domain,
              tier: profile.tier,
              locality: profile.locality,
            },
          }),
        });
        if (res.ok) socratic = await res.json();
      } catch (err) {}

      if (!socratic) {
        socratic = {
          preamble: 'Understood.',
          critique: 'You addressed the high-level concept, but let us look at the operational risk.',
          followUpProbe: 'How would you measure margin sensitivity if your customer churn unexpectedly doubles next quarter?',
          substanceScore: 6,
        };
      }

      logRoundTelemetry(rawText, socratic.substanceScore || 6, wpm, fillers, blob);

      const feedback = `${socratic.preamble} ${socratic.critique} ${socratic.followUpProbe}`;
      await speakText(feedback);

      advanceNextRound();
    }

    function logRoundTelemetry(transcript, score, wpm, fillers, blob) {
      const q = state.selectedQuestions[state.currentRoundIndex];
      state.roundsData.push({
        round: state.currentRoundIndex + 1,
        question: q.question,
        curatedBy: q.curatedBy,
        driveSource: q.driveSource,
        locality: q.locality,
        curatorIntent: q.debrief.curatorIntent,
        offerHolderAnswer: q.debrief.offerHolderAnswer,
        scratchpadImage: state.currentScratchpadImage,
        transcript,
        score,
        wpm,
        fillers,
        blob,
      });
    }

    async function renderFinalDossier() {
      document.getElementById('view-interview').style.display = 'none';
      document.getElementById('view-dossier').style.display = 'block';

      const attempted = state.roundsData.filter((r) => r.transcript !== '[UNATTEMPTED - SILENCE]' && r.transcript !== '[QUESTION PASSED BY CANDIDATE]');
      const avgScore = state.roundsData.reduce((acc, r) => acc + r.score, 0) / state.roundsData.length;
      const avgWpm = attempted.length > 0 ? Math.round(attempted.reduce((acc, r) => acc + r.wpm, 0) / attempted.length) : 0;
      const totalFillers = state.roundsData.reduce((acc, r) => acc + r.fillers, 0);

      const priorHistory = loadHistoricalScores();
      saveHistoricalScore(Number(avgScore.toFixed(1)));

      let progressText = `🎯 Baseline Session Logged for ${profile.name}`;
      if (priorHistory.length > 0) {
        const lastScore = priorHistory[priorHistory.length - 1].score;
        const delta = (avgScore - lastScore).toFixed(1);
        if (delta >= 0) {
          progressText = `📈 Longitudinal Growth: +${delta} Improvement vs Previous Session (${lastScore}/10 → ${avgScore.toFixed(1)}/10)`;
        } else {
          progressText = `📉 Longitudinal Watch: ${delta} Variance vs Previous Session (${lastScore}/10 → ${avgScore.toFixed(1)}/10)`;
        }
      }
      document.getElementById('dossier-progress-pill').textContent = progressText;

      let readiness = 'Tier 3: Unprepared';
      if (avgScore >= 7.5) readiness = 'Tier 1: High Conviction';
      else if (avgScore >= 5.0) readiness = 'Tier 2: Conditional Hire';

      document.getElementById('kpi-tier').textContent = readiness;
      document.getElementById('kpi-score').textContent = `${avgScore.toFixed(1)} / 10`;

            action: 'report',
